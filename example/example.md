# cff example

Generate a feature named `owl` with a Cubit or a BLoC:

```
dart run cff owl --state cubit
dart run cff owl --state bloc
```

Both variants generate the same `data` and `domain` layers (see the
[generated tree](https://pub.dev/packages/cff#generated-tree)). They differ in
`presentation/bloc/`, the page and `owl_dependencies.dart`.

The files below are generated by `cff` for a project named `my_app`.

## Cubit: `presentation/bloc/owl_cubit.dart`

```dart
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:my_app/core/resources/data_state.dart';
import 'package:my_app/feature/owl/domain/entities/owl.dart';
import 'package:my_app/feature/owl/domain/usecases/delete_owl.dart';
import 'package:my_app/feature/owl/domain/usecases/edit_owl.dart';
import 'package:my_app/feature/owl/domain/usecases/get_owl.dart';
import 'package:my_app/feature/owl/domain/usecases/post_owl.dart';
import 'package:my_app/feature/owl/presentation/bloc/owl_state.dart';

class OwlCubit extends Cubit<OwlState> {
  final GetOwlUseCase _getOwlUseCase;
  final PostOwlUseCase _postOwlUseCase;
  final EditOwlUseCase _editOwlUseCase;
  final DeleteOwlUseCase _deleteOwlUseCase;

  /// Incremented by every request; a response is dropped if a newer request
  /// started meanwhile (e.g. a fetch finishing after a delete).
  int _generation = 0;

  OwlCubit({
    required GetOwlUseCase getOwlUseCase,
    required PostOwlUseCase postOwlUseCase,
    required EditOwlUseCase editOwlUseCase,
    required DeleteOwlUseCase deleteOwlUseCase,
  }) : _getOwlUseCase = getOwlUseCase,
       _postOwlUseCase = postOwlUseCase,
       _editOwlUseCase = editOwlUseCase,
       _deleteOwlUseCase = deleteOwlUseCase,
       super(const OwlInitial());

  Future<void> fetch() {
    return _run(
      OwlOperation.fetch,
      () => _getOwlUseCase(),
      OwlFetchSuccess.new,
    );
  }

  Future<void> post(OwlEntity owlEntity) {
    return _run(
      OwlOperation.post,
      () => _postOwlUseCase(params: owlEntity),
      OwlPostSuccess.new,
    );
  }

  Future<void> edit(String id, OwlEntity owlEntity) {
    return _run(
      OwlOperation.edit,
      () => _editOwlUseCase(
        params: EditOwlParams(id: id, owlEntity: owlEntity),
      ),
      OwlEditSuccess.new,
    );
  }

  Future<void> delete(String id) {
    return _run(
      OwlOperation.delete,
      () => _deleteOwlUseCase(params: id),
      OwlDeleteSuccess.new,
    );
  }

  Future<void> _run<T>(
    OwlOperation operation,
    Future<DataState<T>> Function() request,
    OwlState Function(T data) onSuccess,
  ) async {
    final generation = ++_generation;
    final previous = state.owlEntity;
    emit(OwlLoading(operation, owlEntity: previous));

    final result = await request();
    if (generation != _generation || isClosed) return;

    final error = result.error;
    if (error != null) {
      emit(OwlFailure(operation, error, owlEntity: previous));
    } else {
      emit(onSuccess(result.data as T));
    }
  }
}
```

## BLoC: `presentation/bloc/owl_bloc.dart`

```dart
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:my_app/core/resources/data_state.dart';
import 'package:my_app/feature/owl/domain/usecases/delete_owl.dart';
import 'package:my_app/feature/owl/domain/usecases/edit_owl.dart';
import 'package:my_app/feature/owl/domain/usecases/get_owl.dart';
import 'package:my_app/feature/owl/domain/usecases/post_owl.dart';
import 'package:my_app/feature/owl/presentation/bloc/owl_event.dart';
import 'package:my_app/feature/owl/presentation/bloc/owl_state.dart';

class OwlBloc extends Bloc<OwlEvent, OwlState> {
  final GetOwlUseCase _getOwlUseCase;
  final PostOwlUseCase _postOwlUseCase;
  final EditOwlUseCase _editOwlUseCase;
  final DeleteOwlUseCase _deleteOwlUseCase;

  /// Incremented by every request; a response is dropped if a newer request
  /// started meanwhile (e.g. a fetch finishing after a delete).
  int _generation = 0;

  OwlBloc({
    required GetOwlUseCase getOwlUseCase,
    required PostOwlUseCase postOwlUseCase,
    required EditOwlUseCase editOwlUseCase,
    required DeleteOwlUseCase deleteOwlUseCase,
  }) : _getOwlUseCase = getOwlUseCase,
       _postOwlUseCase = postOwlUseCase,
       _editOwlUseCase = editOwlUseCase,
       _deleteOwlUseCase = deleteOwlUseCase,
       super(const OwlInitial()) {
    on<OwlFetchRequested>((event, emit) {
      return _run(
        emit,
        OwlOperation.fetch,
        () => _getOwlUseCase(),
        OwlFetchSuccess.new,
      );
    });
    on<OwlPostRequested>((event, emit) {
      return _run(
        emit,
        OwlOperation.post,
        () => _postOwlUseCase(params: event.owlEntity),
        OwlPostSuccess.new,
      );
    });
    on<OwlEditRequested>((event, emit) {
      return _run(
        emit,
        OwlOperation.edit,
        () => _editOwlUseCase(
          params: EditOwlParams(id: event.id, owlEntity: event.owlEntity),
        ),
        OwlEditSuccess.new,
      );
    });
    on<OwlDeleteRequested>((event, emit) {
      return _run(
        emit,
        OwlOperation.delete,
        () => _deleteOwlUseCase(params: event.id),
        OwlDeleteSuccess.new,
      );
    });
  }

  Future<void> _run<T>(
    Emitter<OwlState> emit,
    OwlOperation operation,
    Future<DataState<T>> Function() request,
    OwlState Function(T data) onSuccess,
  ) async {
    final generation = ++_generation;
    final previous = state.owlEntity;
    emit(OwlLoading(operation, owlEntity: previous));

    final result = await request();
    if (generation != _generation || emit.isDone) return;

    final error = result.error;
    if (error != null) {
      emit(OwlFailure(operation, error, owlEntity: previous));
    } else {
      emit(onSuccess(result.data as T));
    }
  }
}
```

## BLoC: `presentation/bloc/owl_event.dart`

```dart
import 'package:equatable/equatable.dart';
import 'package:my_app/feature/owl/domain/entities/owl.dart';

sealed class OwlEvent extends Equatable {
  const OwlEvent();

  @override
  List<Object?> get props => [];
}

class OwlFetchRequested extends OwlEvent {
  const OwlFetchRequested();
}

class OwlPostRequested extends OwlEvent {
  final OwlEntity owlEntity;

  const OwlPostRequested(this.owlEntity);

  @override
  List<Object?> get props => [owlEntity];
}

class OwlEditRequested extends OwlEvent {
  final String id;
  final OwlEntity owlEntity;

  const OwlEditRequested(this.id, this.owlEntity);

  @override
  List<Object?> get props => [id, owlEntity];
}

class OwlDeleteRequested extends OwlEvent {
  final String id;

  const OwlDeleteRequested(this.id);

  @override
  List<Object?> get props => [id];
}
```

## Shared by both: `presentation/bloc/owl_state.dart`

```dart
import 'package:dio/dio.dart';
import 'package:equatable/equatable.dart';
import 'package:my_app/feature/owl/domain/entities/owl.dart';
import 'package:my_app/feature/owl/domain/entities/owl_delete_response.dart';

enum OwlOperation { fetch, post, edit, delete }

sealed class OwlState extends Equatable {
  /// The owl currently displayed, kept while loading and on failure.
  final OwlEntity? owlEntity;

  const OwlState({this.owlEntity});

  @override
  List<Object?> get props => [owlEntity];
}

class OwlInitial extends OwlState {
  const OwlInitial();
}

class OwlLoading extends OwlState {
  final OwlOperation operation;

  const OwlLoading(this.operation, {super.owlEntity});

  @override
  List<Object?> get props => [operation, owlEntity];
}

class OwlFetchSuccess extends OwlState {
  const OwlFetchSuccess(OwlEntity owlEntity) : super(owlEntity: owlEntity);
}

class OwlPostSuccess extends OwlState {
  const OwlPostSuccess(OwlEntity owlEntity) : super(owlEntity: owlEntity);
}

class OwlEditSuccess extends OwlState {
  const OwlEditSuccess(OwlEntity owlEntity) : super(owlEntity: owlEntity);
}

class OwlDeleteSuccess extends OwlState {
  final OwlDeleteResponseEntity response;

  const OwlDeleteSuccess(this.response);

  @override
  List<Object?> get props => [response, owlEntity];
}

class OwlFailure extends OwlState {
  final OwlOperation operation;
  final DioException error;

  const OwlFailure(this.operation, this.error, {super.owlEntity});

  @override
  List<Object?> get props => [operation, error, owlEntity];
}
```
